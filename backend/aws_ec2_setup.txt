# PASTE the below script in User Data when creating an EC2 instance

#!/bin/bash
# 1. Update packages
dnf update -y

# 2. Install and start Docker
dnf install -y docker
systemctl start docker
systemctl enable docker

# 3. Install Docker Compose plugin
mkdir -p /usr/local/lib/docker/cli-plugins
curl -SL https://github.com/docker/compose/releases/latest/download/docker-compose-linux-$(uname -m) -o /usr/local/lib/docker/cli-plugins/docker-compose
chmod +x /usr/local/lib/docker/cli-plugins/docker-compose

# 4. Create application directory
mkdir -p /app
cd /app

# 5. Write nginx.conf inline
cat << 'EOF' > nginx.conf
# Rate limit zone: tracks client IP addresses ($binary_remote_addr)
# 10 megabyte zone 'mylimit' can hold ~160,000 IPs
# Limit is set to 1 requests per second
limit_req_zone $binary_remote_addr zone=mylimit:10m rate=1r/s;

server {
    listen 80;
    server_name localhost;

    # Limit uploads to 20MB (necessary for larger TenSEAL context and vector uploads)
    client_max_body_size 20M;

    # Buffer client request bodies to disk/memory before passing to Python/FastAPI
    client_body_buffer_size 10M;

    # Custom message when an upload exceeds client_max_body_size (413)
    error_page 413 @too_large;
    location @too_large {
        default_type application/json;
        return 413 '{"detail": "Upload exceeds the 20MB limit. Check if you are setting up lean TenSEAL parameters!"}';
    }

    location / {
        # Apply the rate limit zone with a burst capacity of 1 request
        # 'nodelay' means requests within burst are processed immediately instead of queued
        limit_req zone=mylimit burst=1 nodelay;
        limit_req_status 429;

        # Forward requests to the internal FastAPI container named 'web' on port 8000
        proxy_pass http://web:8000;
        
        # Standard proxy headers to preserve original client request metadata
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
}
EOF

# 6. Write docker-compose.yml inline
# Note: Uses the pre-built image from Docker Hub instead of building on the t3.micro.
cat << 'EOF' > docker-compose.yml
version: '3.8'

services:
  # The FastAPI Python server
  web:
    image: danmurat/priv-mail-server:latest
    expose:
      - "8000"  # Expose port 8000 only inside the Docker virtual network
    restart: unless-stopped
    deploy:
      resources:
        limits:
          memory: 750M # Prevent memory leaks or heavy loads from freezing the 1GB VM

  # The Nginx reverse proxy gatekeeper
  nginx:
    image: nginx:alpine
    ports:
      - "80:80"  # Map host port 80 to Nginx container port 80
    volumes:
      - ./nginx.conf:/etc/nginx/conf.d/default.conf:ro  # Mount custom config
    depends_on:
      - web
    restart: unless-stopped
EOF

# 7. Start the application
docker compose up -d
